function setupTabs() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  [[TAB_PRICES, HEAD_PRICES], [TAB_OIL, HEAD_OIL],
   [TAB_REG, HEAD_REG], [TAB_NEW, HEAD_NEW], [TAB_SRC, HEAD_SRC],
   [TAB_WEB, HEAD_WEB], [TAB_PEND, HEAD_PEND]].forEach(([name, head]) => {
    let sh = ss.getSheetByName(name);
    if (!sh) sh = ss.insertSheet(name);
    if (sh.getLastRow() === 0) {
      sh.getRange(1, 1, 1, head.length).setValues([head])
        .setFontWeight('bold').setBackground('#1B2A4A').setFontColor('#FFFFFF');
      sh.setFrozenRows(1);
    }
  });
  migrateNewItems_();
}

/** ย้ายโครงแท็บ NEW_ITEMS เก่า (ไม่มีคอลัมน์หน่วย) → ใส่คอลัมน์หน่วยให้ พร้อมเติมหน่วยย้อนหลังจาก PRICES */
function migrateNewItems_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const nw = ss.getSheetByName(TAB_NEW);
  if (!nw || nw.getLastColumn() < 2) return;
  if (String(nw.getRange(1, 2).getValue()).trim() === 'หน่วย') return;   // ย้ายแล้ว
  nw.insertColumnAfter(1);
  nw.getRange(1, 1, 1, HEAD_NEW.length).setValues([HEAD_NEW])
    .setFontWeight('bold').setBackground('#1B2A4A').setFontColor('#FFFFFF');
  const n = nw.getLastRow() - 1;
  if (n > 0) {
    const unitOf = {};
    const pr = ss.getSheetByName(TAB_PRICES);
    if (pr && pr.getLastRow() > 1)
      pr.getRange(2, 3, pr.getLastRow() - 1, 2).getValues().forEach(function (r) {
        const k = normName_(r[0]); if (k && !unitOf[k]) unitOf[k] = r[1];
      });
    const names = nw.getRange(2, 1, n, 1).getValues();
    nw.getRange(2, 2, n, 1).setValues(names.map(function (r) {
      return [unitOf[normName_(r[0])] || ''];
    }));
  }
  Logger.log('ย้ายโครง NEW_ITEMS + เติมหน่วยย้อนหลังแล้ว');
}

function doGet(e) {
  const p = (e && e.parameter) || {};
  /* ช่องดีบั๊กน้ำมัน (รุ่น 12): เปิดผ่านเบราว์เซอร์ได้เลย
     …/exec?fn=oiltest&token=รหัส&prov=ขอนแก่น   → ลองแกะราคา + ถ้าไม่ได้ แนบคำตอบดิบจาก ปตท.
     …/exec?fn=oilraw&token=รหัส&prov=ขอนแก่น    → คำตอบดิบทั้งก้อน */
  if (p.token === TOKEN && p.fn === 'oilraw') {
    return ContentService.createTextOutput(rawOilResponse(p.prov || OIL_PROVINCES[0]))
      .setMimeType(ContentService.MimeType.TEXT);
  }
  /* รุ่น 18: …?fn=oilmethods → ดูว่าบริการของ ปตท. มีเมธอดอะไรให้เรียกบ้าง (ไว้ไล่ปัญหา กทม.) */
  if (p.token === TOKEN && p.fn === 'oilmethods') {
    const res = UrlFetchApp.fetch(OIL_URL, { muteHttpExceptions: true });
    const ops = (res.getContentText().match(/\.asmx\?op=([A-Za-z0-9_]+)/g) || [])
                  .map(function (s) { return s.split('=')[1]; });
    return ContentService.createTextOutput(JSON.stringify(
      { ok: true, methods: [].concat(ops).filter(function (v, i, a) { return a.indexOf(v) === i; }) }, null, 1))
      .setMimeType(ContentService.MimeType.JSON);
  }
  /* รุ่น 18: …?fn=oilcall&m=ชื่อเมธอด&args=<Language>th</Language> → ยิงเมธอดใดก็ได้ ดูคำตอบดิบ */
  if (p.token === TOKEN && p.fn === 'oilcall') {
    return ContentService.createTextOutput(
      soapCall_(p.m || 'CurrentOilPrice', p.args || '<Language>th</Language>').slice(0, 40000))
      .setMimeType(ContentService.MimeType.TEXT);
  }
  /* รุ่น 18: …?fn=oildist → ดูราคาดีเซลครบทุกอำเภอที่ ปตท. ส่งมา (ไว้เทียบกับหน้าเว็บ) */
  if (p.token === TOKEN && p.fn === 'oildist') {
    const prv = p.prov || OIL_PROVINCES[0];
    const raw = rawOilResponse(prv);
    const mm = raw.match(/<CurrentOilPriceProvincialResult>([\s\S]*?)<\/CurrentOilPriceProvincialResult>/);
    const rows = parseOilRows_(unesc(mm ? mm[1] : raw));
    const ds = rows.filter(function (r) { return /ดีเซล|diesel/i.test(r.product); })
                   .map(function (r) { return { อำเภอ: r.loc, ชนิด: r.product, ราคา: r.price, มีผล: r.date }; });
    return ContentService.createTextOutput(JSON.stringify(
      { ok: true, prov: prv, ทั้งหมด: rows.length,
        ที่เลือกใช้: pickDieselRow_(rows, prv), ดีเซล: ds }, null, 1))
      .setMimeType(ContentService.MimeType.JSON);
  }
  if (p.token === TOKEN && p.fn === 'oiltest') {
    const prov = p.prov || OIL_PROVINCES[0];
    let out;
    try { const i = fetchPttorDieselInfo(prov);
          out = { ok: true, prov: prov, price: i.price, location: i.location, ref: i.ref,
                  date: i.date, dateISO: i.dateISO, rows: i.rows }; }
    catch (err) {
      out = { ok: false, prov: prov, error: String(err),
              raw: rawOilResponse(prov).slice(0, 1200) };
    }
    return ContentService.createTextOutput(JSON.stringify(out))
      .setMimeType(ContentService.MimeType.JSON);
  }
  return ContentService.createTextOutput(
    JSON.stringify({ ok: true, ver: 19, service: 'คลังข้อมูลวัสดุก่อสร้าง ขทช.ขอนแก่น' })
  ).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const out = { ok: false };
  try {
    const body = JSON.parse(e.postData.contents);
    if (body.token !== TOKEN) throw new Error('token ไม่ถูกต้อง');
    setupTabs();
    if (body.action === 'oil')           out.added = addOil(body.rows || []);
    else if (body.action === 'oil_now')  {           /* รุ่น 11: ดึงราคาสดจาก ปตท. ตามจังหวัดที่ขอ แล้วบันทึกเข้า OIL ให้เลย */
      const prov = String(body.prov || OIL_PROVINCES[0] || 'ขอนแก่น');
      const info = fetchPttorDieselInfo(prov);       /* รุ่น 18: ราคา ณ อำเภอเมือง + วันที่มีผลจริง */
      const n = addOil([[info.date, prov, info.price, oilNote_(info, 'ปุ่มแดชบอร์ด')]]);
      out.price = info.price; out.prov = prov; out.date = info.date;
      out.location = info.location; out.ref = info.ref; out.added = n;
    }
    else if (body.action === 'submit')   {           /* รุ่น 17: ส่งเข้าคิวรออนุมัติ + แจ้งอีเมลผู้อนุมัติ */
      const r = submitPending(body.rows || [], body.by || '');
      out.batch = r.batch; out.added = r.added; out.mailed = r.mailed;
    }
    else if (body.action === 'approve')  out.result = decidePending(body.pass, body.batch, true);
    else if (body.action === 'reject')   out.result = decidePending(body.pass, body.batch, false);
    else if (body.action === 'registry') out.added = setRegistry(body.rows || [], body.mode);
    else if (body.action === 'sources')  out.added = setSources(body.rows || [], body.mode);
    else if (body.action === 'clear')    { const r = clearScope(body.scope, body.opt || {});
                                           out.removed = r.removed; out.backup = r.backup; out.added = 0; }
    else                                 out.added = addPrices(body.rows || []);
    out.ok = true;
  } catch (err) {
    out.error = String(err);
  }
  return ContentService.createTextOutput(JSON.stringify(out))
    .setMimeType(ContentService.MimeType.JSON);
}

/** ส่งชุดราคาเข้าคิวรออนุมัติ + อีเมลแจ้งผู้อนุมัติ — ยังไม่แตะแท็บ PRICES */
function submitPending(rows, by) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(TAB_PEND);
  const now = new Date();
  const batch = 'B' + Utilities.formatDate(now, 'Asia/Bangkok', 'yyMMdd-HHmmss');
  const stamp = Utilities.formatDate(now, 'Asia/Bangkok', 'dd/MM/') +
                (now.getFullYear() + 543) + ' ' + Utilities.formatDate(now, 'Asia/Bangkok', 'HH:mm');
  const who = String(by || '').trim() || '(ไม่ระบุชื่อ)';
  const fresh = rows.map(function (r) {
    return [batch, r[0], r[1], r[2], r[3], r[4], r[5], r[6] || '', r[7] || '', who, stamp, PEND_WAIT, ''];
  });
  if (!fresh.length) return { batch: batch, added: 0, mailed: false };
  sh.getRange(sh.getLastRow() + 1, 1, fresh.length, HEAD_PEND.length).setValues(fresh);

  /* อีเมลแจ้งเตือน — ส่งไม่ผ่านก็ไม่ทำให้การส่งเข้าคิวล้มเหลว */
  let mailed = false;
  try {
    const list = fresh.slice(0, 15).map(function (r) {
      return '• ' + r[3] + '  ' + Number(r[5]).toLocaleString() + ' บาท/' + r[4] + '  — ' + r[6];
    }).join('\n');
    const more = fresh.length > 15 ? ('\n… และอีก ' + (fresh.length - 15) + ' รายการ') : '';
    MailApp.sendEmail(ADMIN_EMAIL,
      '[รออนุมัติ] มีราคาวัสดุ ' + fresh.length + ' รายการรอขึ้นคลัง · ' + batch,
      'มีผู้ส่งราคาเข้าคิวรออนุมัติในระบบฐานข้อมูลวัสดุ\n\n' +
      'ผู้กรอก : ' + who + '\n' +
      'เมื่อ    : ' + stamp + '\n' +
      'รหัสชุด : ' + batch + '\n' +
      'จำนวน   : ' + fresh.length + ' รายการ\n\n' +
      'รายการ (สูงสุด 15 แถวแรก)\n' + list + more + '\n\n' +
      'ราคายังไม่เข้าคลัง PRICES จนกว่าท่านจะอนุมัติ\n' +
      'วิธีอนุมัติ: เปิดแดชบอร์ด → หน้า นำเข้าข้อมูล → การ์ด "รายการรออนุมัติ" → ใส่รหัสผู้ดูแล → กดอนุมัติ\n' +
      (SHEET_LINK ? ('\nชีทคลังข้อมูล: ' + SHEET_LINK + '\n') : ''));
    mailed = true;
  } catch (e) {
    Logger.log('ส่งอีเมลไม่สำเร็จ: ' + e);
  }
  return { batch: batch, added: fresh.length, mailed: mailed };
}

/** อนุมัติ (ok=true) หรือไม่อนุมัติ (ok=false) ทั้งชุด — ตรวจรหัสผู้ดูแลฝั่งเซิร์ฟเวอร์ */
function decidePending(pass, batch, ok) {
  if (String(pass) !== ADMIN_PASS) throw new Error('รหัสผู้ดูแลไม่ถูกต้อง');
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(TAB_PEND);
  if (!sh || sh.getLastRow() < 2) return { moved: 0, marked: 0 };
  const n = sh.getLastRow() - 1;
  const vals = sh.getRange(2, 1, n, HEAD_PEND.length).getValues();
  const now = new Date();
  const stamp = Utilities.formatDate(now, 'Asia/Bangkok', 'dd/MM/') +
                (now.getFullYear() + 543) + ' ' + Utilities.formatDate(now, 'Asia/Bangkok', 'HH:mm');
  const take = [];
  vals.forEach(function (r, i) {
    if (String(r[0]) !== String(batch) || String(r[11]) !== PEND_WAIT) return;
    if (ok) take.push([r[1], r[2], r[3], r[4], r[5], r[6], r[7], r[8]]);
    vals[i][11] = ok ? 'อนุมัติแล้ว' : 'ไม่อนุมัติ';
    vals[i][12] = stamp;
  });
  let moved = 0;
  if (ok && take.length) moved = addPrices(take);      /* เข้าคลังจริงตรงนี้เท่านั้น */
  sh.getRange(2, 1, n, HEAD_PEND.length).setValues(vals);
  return { moved: moved, marked: take.length || vals.filter(function (r) {
    return String(r[0]) === String(batch) && String(r[11]) === 'ไม่อนุมัติ'; }).length };
}

/** rows: [ปีพ.ศ., เดือน, รายการวัสดุ, หน่วย, ราคา, แหล่งวัสดุ, ประเภท, ช่องทาง] */
function addPrices(rows) {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(TAB_PRICES);
  const seen = {};
  if (sh.getLastRow() > 1) {
    sh.getRange(2, 1, sh.getLastRow() - 1, 6).getValues().forEach(r => {
      seen[[r[0], r[1], r[2], r[5]].join('|')] = 1;   // ปี|เดือน|รายการ|แหล่ง
    });
  }
  const now = new Date();
  const fresh = [];
  rows.forEach(r => {
    const key = [r[0], r[1], r[2], r[5]].join('|');
    if (seen[key]) return;
    seen[key] = 1;
    fresh.push([r[0], r[1], r[2], r[3], r[4], r[5], r[6] || '', r[7] || '', now]);
  });
  if (fresh.length)
    sh.getRange(sh.getLastRow() + 1, 1, fresh.length, HEAD_PRICES.length).setValues(fresh);
  logNewNames(rows);
  return fresh.length;
}

/** ป้ายช่องทางที่จะเขียนลง NEW_ITEMS — ช่องทางจริงของแถวนั้น + แหล่งที่ให้ราคา
 *  rows: [ปี, เดือน, ชื่อ, หน่วย, ราคา, แหล่ง, ประเภท, ช่องทาง] */
function chanLabel(r) {
  const ch  = String(r[7] || '').split(' · ')[0].trim() || 'ไม่ระบุช่องทาง';
  const src = String(r[5] || '').trim();
  return src ? (ch + ' · ' + src) : ch;
}

/* ───────── การเทียบชื่อกับบัญชีคุม ─────────
   ชื่อจากแหล่งต่างๆ มักต่างกันแค่ช่องว่าง/วรรณยุกต์ จึงทำความสะอาดก่อนเทียบ
   ลำดับการเทียบ: ตรงเป๊ะ → ยุบช่องว่าง → ตัดช่องว่าง+วรรณยุกต์ (หลวมสุด)   */

function normName_(s) {
  return String(s == null ? '' : s)
    .replace(/[\u00A0\u200B\u200C\u200D\uFEFF]/g, ' ')   // NBSP / zero-width
    .replace(/\s+/g, ' ')
    .trim();
}

function tightName_(s) {
  return normName_(s).toLowerCase()
    .replace(/\s/g, '')
    .replace(/[\u0E47-\u0E4E]/g, '')                        // วรรณยุกต์ + ทัณฑฆาต
    .replace(/ท/g, 'ต').replace(/ฏ/g, 'ต').replace(/ฑ/g, 'ด')
    .replace(/ซ/g, 'ส').replace(/ณ/g, 'น').replace(/ญ/g, 'ย')
    .replace(/[ิี]/g, 'ิ').replace(/[ุู]/g, 'ุ');
}

/** ดัชนีชื่อจากบัญชีคุม 3 ระดับ + ถังไว้หาชื่อใกล้เคียง */
/* ─── ชั้นจับคู่ที่ 4 (รุ่น 14): "ลายเซ็นสเปก" ───
   TPSO ชอบแก้ชื่อเล็กๆ น้อยๆ (เปลี่ยนยี่ห้อ · ปีมอก. · รุ่น · สี · รูปแบบตัวเลข/หน่วย)
   ชั้นนี้ตัดส่วนที่ไม่ใช่สเปกออกก่อนเทียบ — ชื่อที่สเปกเดิมแต่เปลือกเปลี่ยนจะยังจับคู่ได้
   ไม่เด้งเข้า NEW_ITEMS ให้ตามจับซ้ำ */
function sigName_(s) {
  s = normName_(s).toLowerCase();
  s = s.replace(/\(?\s*มอก\.?\s*[\d\/\-]+\s*\)?/g, ' ');        // เลข มอก. + ปี
  s = s.replace(/รุ่น\s*\S+/g, ' ');                             // รุ่น XXX
  s = s.replace(/ตรา\s*\S+(\s*\([^)]*\))?/g, ' ');               // ตรา + ชื่อยี่ห้อ 1 คำ (+วงเล็บรุ่น)
  s = s.replace(/ยี่ห้อ\s*\S+/g, ' ');
  s = s.replace(/(\s)สี\S+/g, '$1');                             // สีของสินค้า (ไม่แตะคำขึ้นต้น "สี…" ของหมวดสี)
  s = s.replace(/(\d),(\d)/g, '$1$2');                           // 1,000 → 1000
  s = s.replace(/([1-9]\d*)\.(\d{3})(?=\D|$)/g, '$1$2');         // 1.000 (จุดแทนจุลภาค) → 1000
  s = s.replace(/(\d)\s*[x×*]\s*(\d)/g, '$1x$2');                // 2 x 9 · 3 * 1 → 2x9
  s = s.replace(/(\d+\.\d*?)0+(?=\D|$)/g, '$1').replace(/(\d)\.(?=\D|$)/g, '$1');  // 0.20→0.2, 5.→5
  s = s.replace(/นิ้ว/g, '"').replace(/เมตร/g, 'ม').replace(/sq\.?mm\.?/g, 'ตรมม');
  s = s.replace(/['′]/g, 'ฟุต');                                   // 6' → 6ฟุต
  s = s.replace(/[็-๎]/g, '');                          // วรรณยุกต์/การันต์
  s = s.replace(/[.\s()\/\-–—ๆฯ]/g, '');                         // จุด ช่องว่าง วงเล็บ ฯลฯ
  return s;
}

function buildKnown_() {
  const reg = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(TAB_REG);
  const K = { exact: {}, norm: {}, tight: {}, sig: {}, bucket: {}, n: 0 };
  if (!reg || reg.getLastRow() < 2) return K;
  reg.getRange(2, REG_NAME_COL, reg.getLastRow() - 1, 1).getValues().forEach(function (r) {
    const raw = String(r[0] == null ? '' : r[0]);
    if (!raw.trim()) return;
    K.exact[raw.trim()] = 1;
    K.norm[normName_(raw)] = 1;
    const t = tightName_(raw);
    K.tight[t] = 1;
    K.sig[sigName_(raw)] = 1;
    const b = t.slice(0, 8);
    (K.bucket[b] = K.bucket[b] || []).push(normName_(raw));
    K.n++;
  });
  return K;
}

function isKnown_(K, name) {
  const raw = String(name == null ? '' : name).trim();
  if (!raw) return true;
  return !!(K.exact[raw] || K.norm[normName_(raw)] || K.tight[tightName_(raw)] || K.sig[sigName_(raw)]);
}

/** หาชื่อในบัญชีคุมที่ใกล้เคียงที่สุด (ไว้เสนอให้ KORN จับคู่) — คืน '' ถ้าไม่ใกล้พอ */
function nearest_(K, name) {
  const t = tightName_(name);
  const cand = K.bucket[t.slice(0, 8)] || [];
  let best = '', score = 0;
  for (var i = 0; i < cand.length && i < 60; i++) {
    const c = tightName_(cand[i]);
    var k = 0;
    while (k < t.length && k < c.length && t.charAt(k) === c.charAt(k)) k++;
    const sc = k / Math.max(t.length, c.length, 1);
    if (sc > score) { score = sc; best = cand[i]; }
  }
  return score >= 0.6 ? best : '';
}

/** ชื่อที่ไม่อยู่ในบัญชีคุม (REGISTRY) → บันทึกลง NEW_ITEMS รอจับคู่ */
function logNewNames(rows) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const K = buildKnown_();
  if (!K.n) return;                         // ยังไม่วางบัญชีคุม — ไม่ต้องเตือน
  const nw = ss.getSheetByName(TAB_NEW);
  const seen = {};
  if (nw.getLastRow() > 1)
    nw.getRange(2, 1, nw.getLastRow() - 1, 1).getValues()
      .forEach(function (r) { seen[normName_(r[0])] = 1; });
  const now = new Date();
  const fresh = [];
  rows.forEach(function (r) {
    const n = String(r[2] == null ? '' : r[2]).trim();
    if (!n) return;
    const key = normName_(n);
    if (seen[key] || isKnown_(K, n)) return;
    seen[key] = 1;
    const near = nearest_(K, n);
    fresh.push([n, String(r[3] == null ? '' : r[3]).trim(), chanLabel(r), now,
                near ? 'รอจับคู่ · ใกล้เคียง: ' + near : 'รอจับคู่']);
  });
  if (fresh.length)
    nw.getRange(nw.getLastRow() + 1, 1, fresh.length, HEAD_NEW.length).setValues(fresh);
}

/** เก็บกวาดแท็บ NEW_ITEMS (รุ่น 16) — เช็คทุกแถวกับบัญชีคุมใหม่ด้วยชั้นจับคู่ปัจจุบัน
 *  (ตรงเป๊ะ → ยุบช่องว่าง → ตัดวรรณยุกต์ → ลายเซ็นสเปก) แถวไหนจริงๆ แล้วรู้จัก ลบออกให้
 *  ใช้หลังอัปโหลดบัญชีคุมชุดใหม่ หรือหลังอัปเดต Code.gs ที่ชั้นจับคู่ฉลาดขึ้น */
function purgeKnownNewItems() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const K = buildKnown_();
  const ui = (function () { try { return SpreadsheetApp.getUi(); } catch (e) { return null; } })();
  if (!K.n) { if (ui) ui.alert('แท็บ REGISTRY ว่าง — อัปโหลดบัญชีคุมก่อน'); return; }
  const nw = ss.getSheetByName(TAB_NEW);
  if (!nw || nw.getLastRow() < 2) { if (ui) ui.alert('NEW_ITEMS ว่างอยู่แล้ว'); return; }
  const n = nw.getLastRow() - 1;
  const vals = nw.getRange(2, 1, n, HEAD_NEW.length).getValues();
  const keep = vals.filter(function (r) { return !isKnown_(K, r[0]); });
  const removed = n - keep.length;
  if (removed > 0) {
    nw.getRange(2, 1, n, HEAD_NEW.length).clearContent();
    if (keep.length)
      nw.getRange(2, 1, keep.length, HEAD_NEW.length).setValues(keep);
  }
  Logger.log('purge NEW_ITEMS: เอาออก ' + removed + ' · เหลือรอจับคู่จริง ' + keep.length);
  if (ui) ui.alert('เก็บกวาดเสร็จ',
    'เอาชื่อที่จับคู่บัญชีคุมได้แล้วออก ' + removed + ' แถว\nเหลือชื่อใหม่จริงที่รอจับคู่ ' + keep.length + ' แถว',
    ui.ButtonSet.OK);
}

/** ตรวจสุขภาพการจับคู่ชื่อ — รันเมื่อสงสัยว่าทำไมชื่อเด้งเข้า NEW_ITEMS
 *  อ่านชื่อทั้งหมดในแท็บ PRICES มาเทียบกับบัญชีคุม แล้วรายงานผล */
function checkMatch() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const K = buildKnown_();
  if (!K.n) { Logger.log('แท็บ REGISTRY ว่าง — อัปโหลดบัญชีคุมก่อน'); toast_('ยังไม่มีบัญชีคุมในชีท'); return; }

  const pr = ss.getSheetByName(TAB_PRICES);
  if (!pr || pr.getLastRow() < 2) { Logger.log('แท็บ PRICES ว่าง'); toast_('แท็บ PRICES ยังไม่มีข้อมูล'); return; }

  const seen = {}, names = [];
  pr.getRange(2, 3, pr.getLastRow() - 1, 1).getValues().forEach(function (r) {
    const n = String(r[0] == null ? '' : r[0]).trim();
    if (n && !seen[n]) { seen[n] = 1; names.push(n); }
  });

  var ex = 0, nm = 0, ti = 0;
  const miss = [];
  names.forEach(function (n) {
    if (K.exact[n]) { ex++; return; }
    if (K.norm[normName_(n)]) { nm++; return; }
    if (K.tight[tightName_(n)]) { ti++; return; }
    miss.push(n);
  });

  Logger.log('════ ตรวจการจับคู่ชื่อกับบัญชีคุม ════');
  Logger.log('บัญชีคุม (REGISTRY): ' + K.n + ' ชื่อ');
  Logger.log('ชื่อไม่ซ้ำในคลัง (PRICES): ' + names.length + ' ชื่อ');
  Logger.log('  ตรงเป๊ะ                 ' + ex);
  Logger.log('  ตรงหลังยุบช่องว่าง       ' + nm + '   ← รุ่นก่อน 10 นับเป็น "ชื่อใหม่" ผิด');
  Logger.log('  ตรงหลังตัดวรรณยุกต์      ' + ti);
  Logger.log('  ไม่พบในบัญชีคุมจริง      ' + miss.length);
  if (miss.length) {
    Logger.log('── ตัวอย่างชื่อที่ยังไม่มีในบัญชีคุม (สูงสุด 20) ──');
    miss.slice(0, 20).forEach(function (n) {
      const near = nearest_(K, n);
      Logger.log('   ' + n + (near ? '   → ใกล้เคียง: ' + near : ''));
    });
  }
  toast_('ตรง ' + (ex + nm + ti) + ' / ' + names.length + ' ชื่อ · ไม่พบจริง ' + miss.length + ' ชื่อ (ดูรายละเอียดใน บันทึกการทำงาน)');
  return { total: names.length, exact: ex, norm: nm, tight: ti, missing: miss.length };
}

/** ซ่อมแท็บ NEW_ITEMS ย้อนหลัง (รันมือครั้งเดียว)
 *  1) แก้ช่องทางที่เคยขึ้น TPSO ผิด → ใช้ช่องทางจริงจากแท็บ PRICES
 *  2) ลบแถวที่จับคู่กับบัญชีคุมเรียบร้อยแล้ว (มีอยู่ใน REGISTRY แล้ว) */
function cleanNewItems() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const nw = ss.getSheetByName(TAB_NEW);
  if (!nw || nw.getLastRow() < 2) { Logger.log('NEW_ITEMS ว่าง ไม่มีอะไรต้องซ่อม'); return; }

  const known = {};
  const reg = ss.getSheetByName(TAB_REG);
  if (reg && reg.getLastRow() > 1)
    reg.getRange(2, REG_NAME_COL, reg.getLastRow() - 1, 1).getValues()
       .forEach(r => { known[String(r[0]).trim()] = 1; });

  const realChan = {}, realUnit = {};
  const pr = ss.getSheetByName(TAB_PRICES);
  if (pr && pr.getLastRow() > 1)
    pr.getRange(2, 1, pr.getLastRow() - 1, 8).getValues()
      .forEach(r => { const n = String(r[2]).trim();
        if (n && !realChan[n]) { realChan[n] = chanLabel(r); realUnit[n] = r[3]; } });

  const K = buildKnown_();                             /* รุ่น 17: เดิมลืมสร้างตัวเทียบ ทำให้ตรวจไม่ออก */
  const vals = nw.getRange(2, 1, nw.getLastRow() - 1, HEAD_NEW.length).getValues();
  const keep = [];
  let fixed = 0, dropped = 0;
  vals.forEach(r => {
    const n = String(r[0]).trim();
    if (!n) return;
    if (K.n && isKnown_(K, n)) { dropped++; return; }   // จับคู่แล้ว — เอาออกจากคิว
    if (!String(r[1]).trim() && realUnit[n]) { r[1] = realUnit[n]; fixed++; }   // เติมหน่วยที่ขาด
    const want = realChan[n];
    if (want && String(r[2]).trim() !== want) { r[2] = want; fixed++; }
    keep.push(r);
  });

  nw.getRange(2, 1, vals.length, HEAD_NEW.length).clearContent();
  if (keep.length) nw.getRange(2, 1, keep.length, HEAD_NEW.length).setValues(keep);
  Logger.log('ซ่อมช่องทาง ' + fixed + ' แถว · เอาชื่อที่จับคู่แล้วออก ' + dropped + ' แถว · เหลือรอจับคู่ ' + keep.length + ' แถว');
}

/** บัญชีคุมจากแดชบอร์ด — mode 'replace' = ล้างแท็บก่อนแล้วเขียนใหม่ทั้งชุด, อื่นๆ = ต่อท้าย
 *  rows: [ลำดับ, ชื่อเดิม (ที่พบจากแหล่ง), หมวด, หน่วย, ชื่อมาตรฐาน] */
function setRegistry(rows, mode) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(TAB_REG);
  if (!sh) sh = ss.insertSheet(TAB_REG);
  if (mode === 'replace') {
    sh.clearContents();
    sh.getRange(1, 1, 1, HEAD_REG.length).setValues([HEAD_REG])
      .setFontWeight('bold').setBackground('#1B2A4A').setFontColor('#FFFFFF');
    sh.setFrozenRows(1);
  }
  if (rows.length) {
    const clean = rows.map(r => [r[0], r[1], r[2], r[3], r[4]]);
    sh.getRange(sh.getLastRow() + 1, 1, clean.length, HEAD_REG.length).setValues(clean);
  }
  return rows.length;
}

/** พิกัดแหล่งวัสดุจากแดชบอร์ด — mode 'replace' = ล้างแท็บก่อนเขียนใหม่ทั้งชุด
 *  rows: [ชื่อแหล่ง, ละติจูด, ลองจิจูด, หมายเหตุ] */
/** mode: 'replace' = ล้างแล้วเขียนใหม่ทั้งชุด (ใช้ตอนอัปโหลดบัญชีคุม)
 *        'upsert'  = ชื่อซ้ำให้ทับพิกัดเดิม ชื่อใหม่ต่อท้าย (รุ่น 19 — ใช้ตอนเติมพิกัดทีละแหล่ง)
 *        อื่น ๆ    = ต่อท้ายอย่างเดียว (พฤติกรรมเดิม) */
function setSources(rows, mode) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(TAB_SRC);
  if (!sh) sh = ss.insertSheet(TAB_SRC);
  if (mode === 'replace') {
    sh.clearContents();
    sh.getRange(1, 1, 1, HEAD_SRC.length).setValues([HEAD_SRC])
      .setFontWeight('bold').setBackground('#1B2A4A').setFontColor('#FFFFFF');
    sh.setFrozenRows(1);
  }
  if (!rows.length) return 0;
  const clean = rows.map(r => [String(r[0]).trim(), r[1], r[2], r[3] || '']);

  if (mode === 'upsert' && sh.getLastRow() > 1) {
    const n = sh.getLastRow() - 1;
    const cur = sh.getRange(2, 1, n, HEAD_SRC.length).getValues();
    const at = {};
    cur.forEach(function (r, i) { at[String(r[0]).trim()] = i; });
    const fresh = [];
    clean.forEach(function (r) {
      const i = at[r[0]];
      if (i === undefined) { fresh.push(r); return; }
      cur[i][1] = r[1]; cur[i][2] = r[2];
      if (r[3]) cur[i][3] = r[3];        /* หมายเหตุว่าง = ไม่ทับของเดิม */
    });
    sh.getRange(2, 1, n, HEAD_SRC.length).setValues(cur);
    if (fresh.length)
      sh.getRange(sh.getLastRow() + 1, 1, fresh.length, HEAD_SRC.length).setValues(fresh);
    return clean.length;
  }

  sh.getRange(sh.getLastRow() + 1, 1, clean.length, HEAD_SRC.length).setValues(clean);
  return clean.length;
}

/** ยิง SOAP ไปที่บริการของ ปตท. เมธอดใดก็ได้ (รุ่น 18 — เดิมยิงได้เมธอดเดียว) */
function soapCall_(method, innerXml) {
  const env = '<?xml version="1.0" encoding="utf-8"?>'
    + '<soap:Envelope xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"'
    + ' xmlns:xsd="http://www.w3.org/2001/XMLSchema"'
    + ' xmlns:soap="http://schemas.xmlsoap.org/soap/envelope/">'
    + '<soap:Body><' + method + ' xmlns="' + OIL_NS + '">'
    + (innerXml || '')
    + '</' + method + '></soap:Body></soap:Envelope>';
  const res = UrlFetchApp.fetch(OIL_URL, {
    method: 'post',
    contentType: 'text/xml; charset=utf-8',
    headers: { SOAPAction: '"https://orapiweb.pttor.com/' + method + '"' },
    payload: env,
    muteHttpExceptions: true
  });
  return 'HTTP ' + res.getResponseCode() + '\n' + res.getContentText();
}

function rawOilResponse(province) {
  return soapCall_('CurrentOilPriceProvincial',
    '<Language>th</Language><Province>' + province + '</Province>');
}

/** ดึงเนื้อในผลลัพธ์ออกจากซองตอบกลับ แล้วถอด &lt; &gt; ให้เป็น XML ปกติ */
function oilBody_(raw) {
  const m = String(raw).match(/<\w*Result>([\s\S]*?)<\/\w*Result>/);
  return unesc(m ? m[1] : raw);
}

/* ── กรุงเทพฯ (รุ่น 18) ────────────────────────────────────────────────
 *  บริการรายจังหวัดของ ปตท. ไม่มีข้อมูล กทม. เลย (ลองชื่อ กรุงเทพมหานคร/กรุงเทพฯ/

 *  "ราคาขายปลีก กทม. และปริมณฑล" ซึ่งจังหวัดปริมณฑลในบริการนี้มีครบ
 *  ตรวจกับหน้าเว็บ ปตท. วันที่ 12 ส.ค. 69 แล้ว: เว็บโชว์ดีเซล 37.54

 *  (นครปฐม/สมุทรสาคร ราคาต่างออกไป จึงไม่เอามาเป็นตัวแทน) */
const OIL_PROXY = {

/** ทางสำรองสุดท้าย — บริการ "ราคาปกติ" (ไม่ระบุจังหวัด)
 *  ลองหลายชื่อเมธอด แล้วจำตัวที่ใช้ได้ไว้ ครั้งต่อไปยิงตรงตัวเดียว
 *  (บริการนี้มีเมธอด CurrentOilPrice / GetOilPrice / *Provincial — ยืนยันจากหน้า WSDL แล้ว) */
const OIL_BASE_METHODS = ['CurrentOilPrice', 'CurrentOilPriceStandard', 'CurrentOilPriceBangkok'];
function fetchPttorBaseRows_() {
  const props = PropertiesService.getScriptProperties();
  const known = props.getProperty('OIL_BASE_METHOD');
  const list = known ? [known].concat(OIL_BASE_METHODS) : OIL_BASE_METHODS;

      if (rows.length) { props.setProperty('OIL_BASE_METHOD', list[i]); return rows; }
    } catch (e) { /* ลองชื่อถัดไป */ }
  }
  return [];
}

/* บางจังหวัด ปตท. ใช้ชื่อไม่ตรงกับชื่อทางการ — ลองทุกชื่อเรียกจนกว่าจะได้ราคา (รุ่น 13) */
const PROV_ALIAS = {
  'กรุงเทพมหานคร': ['กรุงเทพมหานคร', 'กรุงเทพฯ', 'กรุงเทพ', 'กทม.'],
  'กทม.':          ['กทม.', 'กรุงเทพฯ', 'กรุงเทพ', 'กรุงเทพมหานคร'],
};

/** คืนแค่ตัวเลขราคา (ของเดิม โค้ดที่อื่นยังเรียกใช้ได้เหมือนเดิม) */

  return fetchPttorDieselInfo(province).price;

/** รุ่น 18 — คืนรายละเอียดครบ {price, location, date, dateISO, rows}
 *  location = อำเภอที่ใช้อ้างอิง (ปกติคือ "เมือง<จังหวัด>" ตาม ว 477)
 *  date     = วันที่ราคามีผล แปลงเป็น dd/MM/พ.ศ. แล้ว (ไม่ใช่วันที่กดดึง) */
function fetchPttorDieselInfo(province) {
  const names = PROV_ALIAS[String(province).trim()] || [province];
  let lastErr = null;
  for (var i = 0; i < names.length; i++) {
    try {
      const raw = rawOilResponse(names[i]);
      const m = raw.match(/<CurrentOilPriceProvincialResult>([\s\S]*?)<\/CurrentOilPriceProvincialResult>/);
      const body = unesc(m ? m[1] : raw);

      /* 1) ทางหลัก — แกะเป็นก้อนรายอำเภอ แล้วเลือกอำเภอเมือง */
      const rows = parseOilRows_(body);
      const hit  = pickDieselRow_(rows, names[i]);
      if (hit) {
        const loc = hit.loc.replace(/^อ\.?\s*/, '');
        return { price: hit.price, location: loc, ref: loc ? 'อ.' + loc : '',
                 date: thDate_(hit.date), dateISO: hit.date, rows: rows.length };
      }

      /* 2) ทางสำรอง — คำตอบผิดรูปแบบ ใช้ตัวแกะแบบข้อความของเดิม */
      const price = pickDiesel(body);
      if (price != null) {
        const d = new Date();
        return { price: price, location: '', ref: '', rows: 0,
                 date: Utilities.formatDate(d, 'Asia/Bangkok', 'dd/MM/') + (d.getFullYear() + 543),
                 dateISO: '' };
      }
      lastErr = new Error('หาราคาดีเซลไม่พบในคำตอบ (ลองชื่อ "' + names[i] + '")');
    } catch (e) { lastErr = e; }
  }

  /* 3) กรุงเทพฯ — บริการรายจังหวัดไม่มี กทม. ให้ยืมราคาจากจังหวัดปริมณฑลที่ประกาศชุดเดียวกัน */
  const proxies = OIL_PROXY[String(province).trim()] || [];
  for (var q = 0; q < proxies.length; q++) {
    try {
      const raw2 = rawOilResponse(proxies[q]);
      const rows2 = parseOilRows_(oilBody_(raw2));
      const hit2  = pickDieselRow_(rows2, proxies[q]);
      if (hit2) return {
        price: hit2.price, location: OIL_PROXY_LABEL,
        ref: OIL_PROXY_LABEL + ' [อ้างอิง จ.' + proxies[q] + ']',
        date: thDate_(hit2.date), dateISO: hit2.date, rows: rows2.length
      };
    } catch (e) { lastErr = e; }
  }

  /* 4) ทางสำรองสุดท้าย → บริการ "ราคาปกติ" */
  try {
    const brows = fetchPttorBaseRows_();
    const bhit  = pickDieselRow_(brows, province);
    if (bhit) {
      const loc = String(bhit.loc || '').replace(/^อ\.?\s*/, '');
      return { price: bhit.price, location: loc || 'กทม.-ปริมณฑล',
               ref: loc ? 'อ.' + loc : 'เขต กทม.-ปริมณฑล',
               date: thDate_(bhit.date), dateISO: bhit.date, rows: brows.length };
    }
  } catch (e) { /* ปล่อยไปแจ้ง error เดิม */ }

  throw lastErr || new Error('ดึงราคาไม่สำเร็จ');
}

/** แกะคำตอบ ปตท. เป็นแถว {loc, product, price, date} ทีละอำเภอ-ทีละชนิดน้ำมัน */
function parseOilRows_(body) {
  const out = [];
  const S = String(body);
  let blk = S.match(/<\s*FUEL_PROVINCIAL(?:\s[^>]*)?>[\s\S]*?<\s*\/\s*FUEL_PROVINCIAL\s*>/gi);
  /* บริการราคาปกติ (กรุงเทพฯ) ใช้ชื่อก้อนอื่น — หาชื่อก้อนเองโดยนับจำนวนแท็ก
     ก้อนที่ห่อข้อมูล 1 แถว ต้องปรากฏเท่ากับจำนวน <PRODUCT> พอดี */
  if (!blk || !blk.length) {
    const cnt = {}; let mm; const re = /<\s*([A-Za-z_][\w.-]*)(?:\s[^>]*)?>/g;
    while ((mm = re.exec(S)) !== null) { const k = mm[1].toUpperCase(); cnt[k] = (cnt[k] || 0) + 1; }
    const need = cnt['PRODUCT'] || 0;
    let name = '';
    if (need > 0) Object.keys(cnt).forEach(function (k) {
      if (!name && cnt[k] === need && ['PRODUCT', 'PRICE', 'LOCATION', 'PRICE_DATE'].indexOf(k) < 0) name = k;
    });
    if (name) {
      const esc = name.replace(/[.*+?^${}()|[\]\\-]/g, '\\$&');
      blk = S.match(new RegExp('<\\s*' + esc + '(?:\\s[^>]*)?>[\\s\\S]*?<\\s*/\\s*' + esc + '\\s*>', 'gi'));
    }
  }
  if (!blk) blk = [];
  /* (?:\s[^>]*)? สำคัญมาก — กันไม่ให้ <PRICE> ไปคว้า <PRICE_DATE> มาด้วย */
  const tag = function (b, name) {
    const m = b.match(new RegExp('<\\s*' + name + '(?:\\s[^>]*)?>([\\s\\S]*?)<\\s*/\\s*' + name + '\\s*>', 'i'));
    return m ? String(m[1]).replace(/<[^>]*>/g, '').trim() : '';
  };
  for (var i = 0; i < blk.length; i++) {
    const b = blk[i];
    const v = parseFloat(String(tag(b, 'PRICE')).replace(/,/g, '').replace(/[^\d.\-]/g, ''));
    out.push({
      loc:     tag(b, 'LOCATION'),
      product: tag(b, 'PRODUCT'),
      price:   isNaN(v) ? null : v,
      date:    tag(b, 'PRICE_DATE')
    });
  }
  return out;
}

/** เลือกแถว "ดีเซลธรรมดา ณ อำเภอเมือง" — ถ้าไม่มีอำเภอเมือง ค่อยไล่ลงไปตามลำดับสำรอง */
function pickDieselRow_(rows, province) {
  const BAD  = /(B ?20|บี ?20|B ?24|พรีเมียม|premium|เกรดพิเศษ|ซูเปอร์|super)/i;
  const NAME = /(ดีเซล|diesel|hsd)/i;
  const cand = rows.filter(function (r) {
    return r.price != null && r.price >= 20 && r.price < 70
        && NAME.test(r.product) && !BAD.test(r.product);
  });
  if (!cand.length) return null;

  /* ชื่อสินค้าที่ "ธรรมดาที่สุด" มาก่อน เช่น ดีเซล / ดีเซลหมุนเร็ว / ดีเซล B7 */
  const plain = cand.filter(function (r) {
    return /^\s*ดีเซล(\s*หมุนเร็ว)?(\s*B\s*7)?\s*$/i.test(r.product);
  });
  const pool = plain.length ? plain : cand;

  const bare = function (s) { return String(s).replace(/^อ\.?\s*/, '').trim(); };
  const prov = String(province).trim().replace(/^จ\.?\s*/, '');

  /* ลำดับการเลือกอำเภอ — ต้องเทียบแบบตรงตัวก่อน ไม่งั้นพลาดง่ายมาก
     นครราชสีมา มีทั้ง "เมืองนครราชสีมา" (ตัวจริง) และ "เมืองยาง" (คนละอำเภอ)
     ร้อยเอ็ด    มีทั้ง "เมืองร้อยเอ็ด"   (ตัวจริง) และ "เมืองสรวง" (คนละอำเภอ) */
  /* 1) "เมือง" + ชื่อจังหวัด แบบเป๊ะ ๆ — ครอบคลุมเกือบทุกจังหวัด */
  let hit = pool.filter(function (r) { return bare(r.loc) === OIL_DISTRICT + prov; });
  /* 2) ชื่อเดียวกับจังหวัด — พระนครศรีอยุธยา ปตท. ส่งมาว่า "พระนครศรีอยุธยา" ไม่มีคำว่าเมือง */
  if (!hit.length) hit = pool.filter(function (r) { return bare(r.loc) === prov; });
  /* 3) ชื่อว่า "เมือง" เฉย ๆ */
  if (!hit.length) hit = pool.filter(function (r) { return bare(r.loc) === OIL_DISTRICT; });
  /* 4) ขึ้นต้นด้วย "เมือง" (เผื่อสะกดต่างจากชื่อจังหวัดที่เราส่งไป) */
  if (!hit.length) hit = pool.filter(function (r) { return bare(r.loc).indexOf(OIL_DISTRICT) === 0; });
  /* 5) ไม่ระบุอำเภอเลย (เช่นบริการที่คืนราคาเดียวทั้งเขต) */
  if (!hit.length) hit = pool.filter(function (r) { return !bare(r.loc); });
  /* 6) จนปัญญา — เอาตัวแรก */
  if (!hit.length) hit = pool;

  return hit[0];
}

/** '2026-08-12T05:00' → '12/08/2569' (ว่าง = ใช้วันนี้) */
function thDate_(iso) {
  const m = String(iso).match(/(\d{4})-(\d{2})-(\d{2})/);
  const d = new Date();
  if (!m) return Utilities.formatDate(d, 'Asia/Bangkok', 'dd/MM/') + (d.getFullYear() + 543);
  let y = parseInt(m[1], 10);
  if (y < 2200) y += 543;                       // ปตท. ส่งเป็น ค.ศ.
  return m[3] + '/' + m[2] + '/' + y;
}

function unesc(t) {
  return String(t).replace(/&lt;/g, '<').replace(/&gt;/g, '>')
    .replace(/&quot;/g, '"').replace(/&#34;/g, '"')
    .replace(/&#39;/g, "'").replace(/&amp;/g, '&');
}

/** หาเลขราคาดีเซลธรรมดา (ไม่เอา B20 / พรีเมียม) จากข้อความรูปแบบใดก็ได้
 *  รุ่น 12: รับทศนิยม 1-3 ตำแหน่ง · ลูกน้ำแทนจุด · มองไกลขึ้นเมื่อราคาอยู่คนละบรรทัด
 *  · กรองเลขให้อยู่ช่วงราคาน้ำมันจริง (20-70) กันจับเลขปี/รหัสผิดตัว */
function pickDiesel(body) {
  const BAD  = /(B ?20|บี ?20|B ?24|พรีเมียม|premium|เกรดพิเศษ|ซูเปอร์พาวเวอร์|super ?power)/i;
  const NAME = /(ดีเซล|diesel|hsd)/i;
  const numOf = function (s) {
    const m = String(s).match(/(\d{2}(?:[.,]\d{1,3})?)/g);
    if (!m) return null;
    for (var i = 0; i < m.length; i++) {
      const v = parseFloat(m[i].replace(',', '.'));
      if (v >= 20 && v < 70) return v;
    }
    return null;
  };
  // 1) แบบ JSON (ผลลัพธ์เป็น JSON string ทั้งก้อน หรือฝังใน result)
  try {
    const j = JSON.parse(body);
    const flat = [];
    (function walk(o) {
      if (o && typeof o === 'object') { Object.keys(o).forEach(function (k) { walk(o[k]); });
        if (!Array.isArray(o)) flat.push(o); }
    })(j);
    for (var i = 0; i < flat.length; i++) {
      const t = JSON.stringify(flat[i]);
      if (NAME.test(t) && !BAD.test(t)) {
        const v = numOf(t);
        if (v != null) return v;
      }
    }
  } catch (e) { /* ไม่ใช่ JSON */ }
  // 2) แบบข้อความ/XML — บรรทัดที่มีคำว่าดีเซล เอาเลขในบรรทัดนั้นก่อน ไม่มีค่อยไล่บรรทัดถัดไป
  const lines = body.split(/[\r\n<>,{}"]+/);
  for (var k = 0; k < lines.length; k++) {
    const L = lines[k];
    if (NAME.test(L) && !BAD.test(L)) {
      const v = numOf(L);
      if (v != null) return v;
      for (var n = k + 1; n < Math.min(k + 8, lines.length); n++) {
        if (NAME.test(lines[n]) || BAD.test(lines[n])) break;   // เจอชื่อเชื้อเพลิงตัวใหม่ก่อน = ราคาไม่ใช่ของดีเซลแล้ว
        const v2 = numOf(lines[n]);
        if (v2 != null) return v2;
      }
    }
  }
  return null;
}

/** rows: [วันที่(ข้อความ เช่น 16/08/2569), จังหวัด, ราคา, ที่มา] */
function addOil(rows) {
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(TAB_OIL);
  const seen = {};
  if (sh.getLastRow() > 1) {
    sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues().forEach(r => {
      seen[[String(r[0]), r[1]].join('|')] = 1;       // วันที่|จังหวัด
    });
  }
  const fresh = [];
  rows.forEach(r => {
    const key = [String(r[0]), r[1]].join('|');
    if (seen[key]) return;
    seen[key] = 1;
    fresh.push([r[0], r[1], r[2], r[3] || '']);
  });
  if (fresh.length)
    sh.getRange(sh.getLastRow() + 1, 1, fresh.length, HEAD_OIL.length).setValues(fresh);
  return fresh.length;
}

/** รุ่น 18 — เก็บกวาดราคาน้ำมันที่รุ่นก่อนดึงมาผิดอำเภอ แล้วดึงใหม่ให้ถูก
 *  ลบเฉพาะแถวที่ "ที่มา" ขึ้นต้นด้วย ปตท. และยังไม่มีคำว่า อ. กำกับ (= ของรุ่นเก่า)
 *  แถวที่กรอกมือไว้ไม่แตะ · สำรองแท็บ OIL ก่อนเสมอ */
function menuOilRefix() {
  const ui = SpreadsheetApp.getUi();
  const sh = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(TAB_OIL);
  if (!sh || sh.getLastRow() < 2) { ui.alert('แท็บ OIL ยังไม่มีข้อมูล'); return; }
  const vals = sh.getRange(2, 1, sh.getLastRow() - 1, HEAD_OIL.length).getValues();
  const kill = [];
  vals.forEach(function (r, i) {
    const src = String(r[3] || '');
    /* รูปแบบใหม่ (รุ่น 18) มีวันที่ที่ไปดึงกำกับในวงเล็บเสมอ — ไม่มีวันที่ = ของรุ่นเก่า */
    if (/^\s*ปตท\./.test(src) && !/\d{2}\/\d{2}\/\d{4}/.test(src)) kill.push(i + 2);
  });
  if (!kill.length) { ui.alert('ไม่พบแถวของรุ่นเก่า', 'ราคาน้ำมันในชีทเป็นของรุ่น 18 อยู่แล้ว', ui.ButtonSet.OK); return; }
  const ans = ui.alert('ซ่อมราคาน้ำมัน',
    'พบราคาที่ดึงด้วยรุ่นเก่า ' + kill.length + ' แถว (อาจเป็นราคาของอำเภออื่น)\n' +
    'จะสำรองแท็บ OIL แล้วลบแถวเหล่านี้ จากนั้นดึงราคาใหม่ให้ตาม ว 477 (อำเภอเมือง)\n\nดำเนินการเลยไหม',
    ui.ButtonSet.YES_NO);
  if (ans !== ui.Button.YES) return;
  backupTab(TAB_OIL);
  kill.sort(function (a, b) { return b - a; }).forEach(function (r) { sh.deleteRow(r); });
  updateOilAuto();
  ui.alert('เรียบร้อย', 'ลบของเก่า ' + kill.length + ' แถว และดึงราคาใหม่แล้ว\n' +

/* ───────── สำรองข้อมูล ───────── */

function stamp_() {
  return Utilities.formatDate(new Date(), 'Asia/Bangkok', 'yyMMdd_HHmm');
}

/** สำเนาแท็บหนึ่งไปเป็นแท็บ "สำรอง_<ชื่อ>_<เวลา>" แล้วลบชุดเก่าที่เกิน BAK_KEEP */
function backupTab(name) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(name);
  if (!sh || sh.getLastRow() < 1) return '';
  const vals = sh.getDataRange().getValues();
  const bakName = BAK_PREFIX + name + '_' + stamp_();
  const old = ss.getSheetByName(bakName);
  if (old) ss.deleteSheet(old);
  const bak = ss.insertSheet(bakName);
  bak.getRange(1, 1, vals.length, vals[0].length).setValues(vals);
  bak.hideSheet();
  // ลบชุดเก่าที่เกินจำนวนที่เก็บ
  const mine = ss.getSheets()
    .map(function (s) { return s.getName(); })
    .filter(function (n) { return n.indexOf(BAK_PREFIX + name + '_') === 0; })
    .sort();
  while (mine.length > BAK_KEEP) {
    ss.deleteSheet(ss.getSheetByName(mine.shift()));
  }
  return bakName;
}

function backupAll() {
  const made = [TAB_PRICES, TAB_OIL, TAB_REG, TAB_NEW, TAB_SRC, TAB_WEB]
    .map(backupTab).filter(String);
  Logger.log('สำรองแล้ว: ' + made.join(' · '));
  return made;
}

/* ───────── ล้างข้อมูล ─────────
   scope: 'prices' ทั้งแท็บ · 'prices_month' {be,mo} · 'prices_channel' {chan}
          'oil' · 'new' · 'registry' · 'sources'
   คืนค่า {removed, backup}                                        */
function clearScope(scope, opt) {
  opt = opt || {};
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const TABS = { prices: TAB_PRICES, prices_month: TAB_PRICES, prices_channel: TAB_PRICES,
                 oil: TAB_OIL, 'new': TAB_NEW, registry: TAB_REG, sources: TAB_SRC };
  const tab = TABS[scope];
  if (!tab) throw new Error('ไม่รู้จักขอบเขต: ' + scope);
  const sh = ss.getSheetByName(tab);
  if (!sh || sh.getLastRow() < 2) return { removed: 0, backup: '' };

  const backup = backupTab(tab);
  const width = sh.getLastColumn();
  const vals = sh.getRange(2, 1, sh.getLastRow() - 1, width).getValues();
  let keep;

  if (scope === 'prices_month') {
    const be = Number(opt.be), mo = Number(opt.mo);
    keep = vals.filter(function (r) { return !(Number(r[0]) === be && Number(r[1]) === mo); });
  } else if (scope === 'prices_channel') {
    const c = String(opt.chan || '').trim();
    keep = vals.filter(function (r) { return String(r[7] || '').split(' · ')[0].trim() !== c; });
  } else {
    keep = [];
  }

  const removed = vals.length - keep.length;
  sh.getRange(2, 1, vals.length, width).clearContent();
  if (keep.length) sh.getRange(2, 1, keep.length, width).setValues(keep);
  Logger.log('ล้าง ' + scope + ' → เอาออก ' + removed + ' แถว (สำรองไว้ที่ ' + backup + ')');
  return { removed: removed, backup: backup };
}

/** ลบแถวซ้ำในแท็บ PRICES (ยึด ปี|เดือน|รายการ|แหล่ง — เก็บแถวแรกไว้) */
function dedupePrices() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sh = ss.getSheetByName(TAB_PRICES);
  if (!sh || sh.getLastRow() < 3) { toast_('ไม่มีแถวพอให้ตรวจ'); return 0; }
  const backup = backupTab(TAB_PRICES);
  const width = sh.getLastColumn();
  const vals = sh.getRange(2, 1, sh.getLastRow() - 1, width).getValues();
  const seen = {}, keep = [];
  vals.forEach(function (r) {
    if (!String(r[2] || '').trim()) return;
    const k = [r[0], r[1], String(r[2]).trim(), String(r[5]).trim()].join('|');
    if (seen[k]) return;
    seen[k] = 1; keep.push(r);
  });
  const removed = vals.length - keep.length;
  sh.getRange(2, 1, vals.length, width).clearContent();
  if (keep.length) sh.getRange(2, 1, keep.length, width).setValues(keep);
  Logger.log('ลบแถวซ้ำ ' + removed + ' แถว (สำรองไว้ที่ ' + backup + ')');
  toast_('ลบแถวซ้ำ ' + removed + ' แถว · เหลือ ' + keep.length + ' แถว');
  return removed;
}

/** กู้คืนแท็บจากไฟล์สำรอง — วางทับข้อมูลปัจจุบันของแท็บต้นทาง */
function restoreBackup(bakName) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const bak = ss.getSheetByName(bakName);
  if (!bak) throw new Error('ไม่พบแท็บสำรองชื่อ ' + bakName);
  const orig = bakName.replace(BAK_PREFIX, '').replace(/_\d{6}_\d{4}$/, '');
  const sh = ss.getSheetByName(orig);
  if (!sh) throw new Error('ไม่พบแท็บปลายทาง ' + orig);
  backupTab(orig);                              // สำรองของปัจจุบันไว้ก่อนอีกชั้น
  const vals = bak.getDataRange().getValues();
  sh.clearContents();
  sh.getRange(1, 1, vals.length, vals[0].length).setValues(vals);
  Logger.log('กู้คืน ' + orig + ' จาก ' + bakName + ' แล้ว ' + (vals.length - 1) + ' แถว');
  return vals.length - 1;
}
